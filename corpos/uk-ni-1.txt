            let isPreviewOpen = false;
            const PREVIEW_REFRESH_INTERVAL_SECONDS = 1;
            const KEEP_OPEN_SECONDS = 3600;

            function togglePreviewCamera() {
                let previewContainer = document.getElementById("previewCameraContainer");
                let button = document.querySelector('.floating-button');

                if (previewContainer.classList.contains("visible")) {
                    // Hide the preview container and show the button
                    previewContainer.classList.remove("visible");
                    previewContainer.classList.add("hidden");
                    button.classList.remove('open');
                    button.classList.remove('hidden');
                } else {
                    // Show the preview container and hide the button
                    previewContainer.classList.remove("hidden");
                    previewContainer.classList.add("visible");
                    button.classList.add('open');
                    setTimeout(() => {
                        button.classList.add('hidden');
                    }, 300); // Adjust the timeout to match the transition duration
                    //isPreviewOpen = true;
                }

                // Update aria properties based on expansion state
                previewContainer.setAttribute("aria-expanded", previewContainer.classList.contains("visible") ? "true" : "false");
                updateCameraPreviews();
            }

			
            function showPreviewCamera( isExpanded ) {
                let previewContainer = document.getElementById("previewCameraContainer");
                let button = document.querySelector('.floating-button');
                if (previewContainer.classList.contains("hidden")) {
                    // Ensure the preview container is always visible
                    previewContainer.classList.remove("hidden");
                    previewContainer.classList.add("visible");
                }
                
                isPreviewOpen = isExpanded;

                // Update aria properties to reflect the container is expanded
                previewContainer.setAttribute("aria-expanded", "true");
                updateCameraPreviews();
            }

            function removeFromPreview(id) {
                let cameraPreviewList = JSON.parse(Cookies.get('cameraPreviewList') ?? null) ?? [];

                cameraPreviewList = cameraPreviewList.filter(camera => camera.id!==id);
                Cookies.set('cameraPreviewList', JSON.stringify(cameraPreviewList), { expires: 7300 });
                updateCameraPreviews();
            }

                let carouselContainer = document.getElementById("previewCameraCarousel");
                carouselContainer.classList.add("hidden");

                const button = element.querySelector('.cameraPreviewAdd');

            // Every x seconds (PREVIEW_REFRESH_INTERVAL_SECONDS), update the preview cameras including enlarged carousel versions) using a cache breaker
            function refreshPreviews() {

                // check if the preview drawer is open, no need to refresh if it is now
                if (isPreviewOpen === true) {
                    // Loop through all of the elements representing a preview
                    let cameraPreviews = [...document.getElementsByClassName("preview-image"), document.getElementById("previewCameraCarouselCurrentImage")];

                    for (let i=0;i<cameraPreviews.length; i++) {
                        let previewImage = cameraPreviews[i];
                        // Remove existing cache breakers from the url
                        let previewImageSrc = previewImage.src.split("#")[0].split("?")[0];
                        if (previewImageSrc) {
                            // amend the image src using a timestamp at the end (cache breaker), this should force a refresh
                            previewImageSrc = `${previewImageSrc}?${Date.now()}`;
                            previewImage.src = previewImageSrc;
                        }
                    }

                    // Auto-close the drawer if the limit exceeded
                    let carouselContainer = document.getElementById("previewCameraContainer");
                    let opened = carouselContainer.getAttribute("date-opened");
                    if (Date.now() - opened > (KEEP_OPEN_SECONDS * 1000)) {
                        //togglePreviewCamera();
                    }
                    
                     
                }
               
               	setTimeout(refreshPreviews, PREVIEW_REFRESH_INTERVAL_SECONDS*1000);

            }

            function openCarousel(id) {
                let cameraPreviewList = JSON.parse(Cookies.get('cameraPreviewList') ?? null) ?? [];

                let carouselCurrentImage = document.getElementById("previewCameraCarouselCurrentImage");
                let carouselContainer = document.getElementById("previewCameraCarousel");
                let carouselTitle = document.getElementById("previewCameraCarouselTitle");
                let newCarouselImageSrc;
                let newCarouselTitle;

                // Get the image url for the id given
                cameraPreviewList.forEach(camera => {
                    if (camera.id === id) {
                        newCarouselImageSrc = camera.imageRoot + '/' + camera.staticFile + '?cache=' + camera.cameraCache;
                        newCarouselTitle = camera.location;
                    }
                });

                // If the passed id isn't a preview image, do nothing and keep the carousel closed
                if (typeof(newCarouselImageSrc) === 'undefined') {
                    return;
                }

                carouselCurrentImage.dataset.id = id;
                carouselCurrentImage.src = newCarouselImageSrc;
                carouselTitle.innerHTML = newCarouselTitle;
                carouselContainer.classList.remove("hidden");
            }

            function carouselNext() {
                let cameraPreviewList = JSON.parse(Cookies.get('cameraPreviewList') ?? null) ?? [];

                let carouselCurrentImage = document.getElementById("previewCameraCarouselCurrentImage");
                let carouselTitle = document.getElementById("previewCameraCarouselTitle");
                let newCarouselImageSrc;

                // Get the index of the current camera being displayed
                let currentDisplayedId = carouselCurrentImage.dataset.id;
                let currentDisplayedIndex = -1;
                for (let i=0; i<cameraPreviewList.length; i++) {
                    if (cameraPreviewList[i].id == currentDisplayedId) {
                        currentDisplayedIndex = i;
                        break;
                    }
                }

                let newDisplayedIndex = currentDisplayedIndex+1;
                // Move to the next image, wrapping around to 0 if out of bounds
                if (newDisplayedIndex>=cameraPreviewList.length) {
                    newDisplayedIndex = 0;
                }

                let newPreviewCamera = cameraPreviewList[newDisplayedIndex];
                let newPreviewCameraSrc = newPreviewCamera.imageRoot + '/' + newPreviewCamera.staticFile + '?cache=' + newPreviewCamera.cameraCache
                carouselCurrentImage.dataset.id = newPreviewCamera.id;
                carouselTitle.innerHTML = newPreviewCamera.location;
                carouselCurrentImage.src = newPreviewCameraSrc;

            }

            function carouselPrevious() {
                let cameraPreviewList = JSON.parse(Cookies.get('cameraPreviewList') ?? null) ?? [];

                let carouselCurrentImage = document.getElementById("previewCameraCarouselCurrentImage");
                let carouselTitle = document.getElementById("previewCameraCarouselTitle");

                for (let i=0; i<cameraPreviewList.length; i++) {

                        currentDisplayedIndex = i;
                        break;
                    }
                }

                let newDisplayedIndex = currentDisplayedIndex-1;
                // Move to the previous image, wrapping around to maximum if out of bounds
                if (newDisplayedIndex<0) {
                    newDisplayedIndex = cameraPreviewList.length-1;
                }

                let newPreviewCamera = cameraPreviewList[newDisplayedIndex];
                let newPreviewCameraSrc = newPreviewCamera.imageRoot + '/' + newPreviewCamera.staticFile + '?cache=' + newPreviewCamera.cameraCache
                carouselCurrentImage.dataset.id = newPreviewCamera.id;
                carouselTitle.innerHTML = newPreviewCamera.location;
                carouselCurrentImage.src = newPreviewCameraSrc;
            }
